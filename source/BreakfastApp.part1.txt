function normalizeRoom(room: Room): Room {
  const people = Math.max(0, Number(room.people) || 0);
  const arrivedCount = Math.max(0, Math.min(people, Number(room.arrivedCount ?? ((room.present || room.departed) ? people : 0)) || 0));
  const departedCount = Math.max(0, Math.min(arrivedCount, Number(room.departedCount ?? (room.departed ? arrivedCount : 0)) || 0));
  return {
    ...room,
    people,
    arrivedCount,
    departedCount,
    present: arrivedCount > departedCount,
    departed: people > 0 && arrivedCount >= people && departedCount >= arrivedCount,
  };
}

function getArrivedCount(room: Room): number {
  return normalizeRoom(room).arrivedCount || 0;
}

function getDepartedCount(room: Room): number {
  return normalizeRoom(room).departedCount || 0;
}

function getRemainingCount(room: Room): number {
  return Math.max(0, (room.people || 0) - getArrivedCount(room));
}

function roomIsPresent(room: Room): boolean {
  const normalized = normalizeRoom(room);
  return normalized.present;
}

function roomIsDeparted(room: Room): boolean {
  const normalized = normalizeRoom(room);
  return Boolean(normalized.departed);
}

function RoomRow({ room, onOpen, language, editMode = false }: { room: Room; onOpen: (room: Room) => void; language: Language; editMode?: boolean }) {
  const occupied = room.guests.length > 0;
  const c = COPY[language];
  const arrivedCount = getArrivedCount(room);
  const remainingCount = getRemainingCount(room);
  const roomPresent = roomIsPresent(room);
  const roomDeparted = roomIsDeparted(room);
  const finishedText = language === "de" ? "Frühstück beendet" : language === "en" ? "Breakfast finished" : "Đã kết thúc bữa sáng";
  const openText = language === "de" ? "Noch offen" : language === "en" ? "Still open" : "Vẫn còn chờ";
  const arrivedSummary = room.people > 0
    ? language === "de"
      ? `${arrivedCount} von ${room.people} da`
      : language === "en"
        ? `${arrivedCount} of ${room.people} here`
        : `${arrivedCount}/${room.people} đã đến`
    : c.present;
  const laterText = remainingCount > 0
    ? language === "de"
      ? `${remainingCount} ${remainingCount === 1 ? "Person kommt später" : "Personen kommen später"}`
      : language === "en"
        ? `${remainingCount} ${remainingCount === 1 ? "person is coming later" : "people are coming later"}`
        : `${remainingCount} ${remainingCount === 1 ? "người sẽ đến sau" : "người sẽ đến sau"}`
    : "";

  return (
    <button
      className={`room-row ${room.included ? "included" : ""} ${roomPresent ? "present" : ""} ${roomDeparted ? "departed" : ""} ${arrivedCount > 0 && remainingCount > 0 ? "partial" : ""}`}
      onClick={() => onOpen(room)}
      aria-label={`Zimmer ${room.room} öffnen`}
    >
      <span className="room-number">{room.room}</span>
      <span className="people"><Users />{occupied ? room.people : ""}</span>
      <span className="guest-names">
        {occupied ? room.guests.map(name => <strong key={name}>{name}</strong>) : <strong className="vacant">{c.notOccupied}</strong>}
        {room.included && <small className="meta-line"><Coffee /> {c.included}</small>}
        {room.table && (roomPresent || roomDeparted) && <small className="meta-line table-meta"><MapPin /> {room.table}</small>}
        {arrivedCount > 0 && remainingCount > 0 && <small className="meta-line status-note"><Users /> {arrivedSummary}</small>}
      </span>
      <span className="room-tail">
        {room.guestInfo?.length ? <span className="guest-info-indicator" title={language === "de" ? "Gastinfo vorhanden" : language === "en" ? "Guest info available" : "Có thông tin khách"} aria-hidden="true"><Info /></span> : null}
        {roomDeparted
          ? <span className="room-state departed"><LogOut /> {c.leave}</span>
          : roomPresent
            ? <span className={`room-state ${remainingCount > 0 ? "open partial" : "present"}`}>{remainingCount > 0 ? arrivedSummary : <><Check /> {arrivedSummary}</>}</span>
            : room.included
              ? <span className="room-state open">{openText}</span>
              : null}
        {editMode && <span className="room-edit-icon" aria-hidden="true"><PencilLine /></span>}
      </span>
    </button>
  );
}
